# 1. Targeted decompilation of the consumer side (jadx, one class at a time)
for c in com.xiaomi.payment.channel.VoucherFragment com.xiaomi.payment.entryData.CameraIntentData \
         com.xiaomi.payment.entryData.CameraIntentData\$a com.xiaomi.payment.entry.a \
         com.xiaomi.payment.entry.IEntry com.xiaomi.payment.o.l com.xiaomi.payment.o.h \
         com.xiaomi.payment.l.c com.xiaomi.payment.l.f com.mipay.common.data.v0 \
         com.mipay.common.data.h0; do
  $WS/tools/jadx/bin/jadx --single-class "$c" --show-bad-code --no-res \
      --single-class-output "$R/sources/consumer/${c}.java" "$APK"
done
